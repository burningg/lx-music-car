cmake_minimum_required(VERSION 3.22)
project(lx_offline_stt LANGUAGES C CXX)

get_filename_component(STT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.stt" ABSOLUTE)
if(NOT EXISTS "${STT_ROOT}/sherpa-onnx-1.12.15/CMakeLists.txt")
  message(FATAL_ERROR "Run npm run prepare:stt before configuring offline STT")
endif()
set(ENV{SHERPA_ONNXRUNTIME_INCLUDE_DIR} "${STT_ROOT}/onnxruntime-1.17.1/headers")
set(ENV{SHERPA_ONNXRUNTIME_LIB_DIR} "${STT_ROOT}/onnxruntime-1.17.1/jni/${ANDROID_ABI}")
# All transitive sources are pinned/hash-checked by the Node preparer, not downloaded by CMake.
set(FETCHCONTENT_SOURCE_DIR_KALDI_NATIVE_FBANK "${STT_ROOT}/deps/fbank" CACHE PATH "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_KISSFFT "${STT_ROOT}/deps/kissfft" CACHE PATH "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_KALDI_DECODER "${STT_ROOT}/deps/kaldi-decoder" CACHE PATH "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_SIMPLE-SENTENCEPIECE "${STT_ROOT}/deps/sentencepiece" CACHE PATH "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_KALDIFST "${STT_ROOT}/deps/kaldifst" CACHE PATH "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_OPENFST "${STT_ROOT}/deps/openfst" CACHE PATH "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_EIGEN "${STT_ROOT}/deps/eigen" CACHE PATH "" FORCE)
add_compile_definitions(EIGEN_MPL2_ONLY)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_JNI ON CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_TTS OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_SPEAKER_DIARIZATION OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_C_API OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_BINARY OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_PYTHON OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_CHECK OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_PORTAUDIO OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_WEBSOCKET OFF CACHE BOOL "" FORCE)
set(SHERPA_ONNX_ENABLE_GPU OFF CACHE BOOL "" FORCE)
# TTS is deliberately off: do not pull espeak-ng/piper/cppinyin into this ASR-only application.
add_subdirectory("${STT_ROOT}/sherpa-onnx-1.12.15" sherpa EXCLUDE_FROM_ALL)
